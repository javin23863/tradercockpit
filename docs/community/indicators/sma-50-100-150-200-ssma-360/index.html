// This Pine Script® code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/

// © AbaddonPL



// @version=6

indicator("SMA 50 / 100 / 150 / 200 / SSMA 360", overlay=true)



// TAGS: Divergence, Relative Strength, Crypto, Cryptocurrency, Crypto Scanner, Crypto Screener, Divergence Scanner, Multi Ticker Scanner, RSI, MACD, Stochastic, OBV, MFI, CCI, Awesome Oscillator, Bullish Divergence, Bearish Divergence, Market Scanner, Crypto Trading, Bitcoin, BTC, Altcoins, Momentum, Confluence, Trading Indicator, Crypto Indicator, Relative Strength Scanner



// ─────────────────────────────────────────────

// USTAWIENIA UŻYTKOWNIKA

// ─────────────────────────────────────────────



// SMA 50

show50   = input.bool(true,  "Show SMA 50")

color50  = input.color(color.blue, "Color SMA 50")

width50  = input.int(1, "Width SMA 50")



// SMA 100

show100  = input.bool(true,  "Show SMA 100")

color100 = input.color(color.orange, "Color SMA 100")

width100 = input.int(2, "Width SMA 100")



// SMA 150

show150  = input.bool(true,  "Show SMA 150")

color150 = input.color(color.green, "Color SMA 150")

width150 = input.int(3, "Width SMA 150")



// SMA 200

show200  = input.bool(true,  "Show SMA 200")

color200 = input.color(color.red, "Color SMA 200")

width200 = input.int(4, "Width SMA 200")



// SSMA 360

show360  = input.bool(true,  "Show SSMA 360")

color360 = input.color(color.purple, "Color SSMA 360")

width360 = input.int(5, "Width SSMA 360")



// ─────────────────────────────────────────────

// OBLICZENIA

// ─────────────────────────────────────────────



sma50  = ta.sma(close, 50)

sma100 = ta.sma(close, 100)

sma150 = ta.sma(close, 150)

sma200 = ta.sma(close, 200)



// Smoothed SMA 360 (SSMA)

ssma360 = ta.sma(ta.sma(close, 360), 360)



// ─────────────────────────────────────────────

// RYSOWANIE

// ─────────────────────────────────────────────



plot(show50  ? sma50  : na, title="SMA 50",  color=color50,  linewidth=width50)

plot(show100 ? sma100 : na, title="SMA 100", color=color100, linewidth=width100)

plot(show150 ? sma150 : na, title="SMA 150", color=color150, linewidth=width150)

plot(show200 ? sma200 : na, title="SMA 200", color=color200, linewidth=width200)

plot(show360 ? ssma360 : na, title="SSMA 360", color=color360, linewidth=width360)
